49// Windows accepts either separator; POSIX has only the forward slash, and a
50// backslash is an ordinary character in a POSIX file name.
51bool isPathSeparator(char c) {
52#if defined(_WIN32)
53 return c == '/' || c == '\\';
54#else
55 return c == '/';

59// True for a path that starts at the root: /dir on POSIX, C:\dir or \\server
60// on Windows.
61bool isAbsolutePath(const std::string& path) {
62 if (path.empty()) {
63 return false;
64 }
65 if (isPathSeparator(path.front())) {
66 return true;
67 }
68#if defined(_WIN32)
69 if (path.size() >= 3 && std::isalpha(static_cast<unsigned char>(path[0])) != 0 && path[1] == ':' &&
70 isPathSeparator(path[2])) {
71 return true;
72 }
73#endif
74 return false;
75}

85// A relative path in the configuration is resolved against the directory of
86// the config file, so a config file can be moved together with what it names.
87std::string resolveAgainstConfig(const std::string& config_path, const std::string& path) {
88 if (path.empty() || isAbsolutePath(path)) {
89 return path;
90 }
91 const size_t slash = lastPathSeparator(config_path);
92 if (slash == std::string::npos) {
93 return path;
94 }
95 return config_path.substr(0, slash + 1) + path;
96}

98// The system prompt may live in a file next to the config. The file wins over
99// the text in the config, so the prompt can be edited without touching the

101bool applySystemPromptFile(const std::string& config_path, config::Config& config, std::string& error) {
102 if (config.agent.system_prompt_path.empty()) {
103 return true;

106 log::detail("System prompt: " + path);
107 std::string prompt;
108 if (!readTextFile(path, prompt)) {
109 error = "Cannot open system prompt: " + path;
110 return false;

112 config.agent.system_prompt = prompt;
113 return true;

116// The model with the lowest priority number wins.
117const config::ModelConfig* selectModel(const std::vector<config::ModelConfig>& models) {
118 if (models.empty()) {
119 return nullptr;
120 }
121 const auto best = std::min_element(models.begin(), models.end(),
122 [](const config::ModelConfig& left, const config::ModelConfig& right) {
123 return left.priority < right.priority;
124 });
125 return &*best;
126}

357int App::runTask(const cli::Invocation& invocation) {
358 std::string error;
359 const std::string& configPath = invocation.configPath;
360 config::Config config;
361 const config::ModelConfig* selected = nullptr;
362 if (!prepareModel(configPath, config, selected)) {
363 return 1;
364 }
365
366 // The task comes from the command line when one was given, otherwise from
367 // standard input.
368 std::string task = invocation.taskFromArgument ? invocation.task : readAll(in_);
369 if (task.empty()) {
370 err_ << cli::usage();
371 return 1;
372 }
373 log::progress("Task: " + task);
374
375 Runtime runtime;
376 if (!buildRuntime(configPath, config, *selected, runtime)) {
377 return 1;
378 }
379
380 // With a session directory the run creates a record before the first model
381 // call and rewrites it after each turn. Without one the run stays one-shot.
382 std::shared_ptr<session::SessionStore> store;
383 std::shared_ptr<session::SessionRecorder> recorder;
384 if (!config.agent.session_dir.empty()) {
385 std::string directory;
386 if (!sessionDirectoryFor(configPath, config, directory)) {
387 return 1;
388 }
389 store = openSessionStore(directory);
390 if (!store) {
391 return 1;
392 }
393
394 session::SessionState state;
395 state.task = task;
396 state.workspace = config.agent.workspace;
397 state.model = selected->model;
398 if (!store->begin(state, error)) {
399 reportError(error);
400 return 1;
401 }
402 log::progress("Session: " + state.id);
403 recorder = std::make_shared<session::SessionRecorder>(store, std::move(state));
404 }
405
406 agent::AgentLoopConfig loopConfig;
407 loopConfig.system_prompt = buildSystemPrompt(config.agent.system_prompt, runtime.fragments);
408 loopConfig.workspace = config.agent.workspace;
409 loopConfig.stop_requested = [] { return platform::shutdownRequested(); };
410 agent::AgentLoop loop(loopConfig, runtime.model, runtime.context, runtime.tools, recorder);
411
412 std::string result;
413 if (!loop.run(task, result, error)) {
414 reportError(error);
415 return 1;
416 }
417 log::progress("Done");
418 out_ << result << "\n";
419 return 0;
420}

422App::ResumeAction App::loadResumeSession(const session::SessionStore& store, const std::string& requested,
423 session::SessionState& state, std::string& error) {
424 std::string id = requested;
425 if (id.empty()) {
426 id = store.latest(error);
427 if (id.empty()) {
428 if (error.empty()) {
429 error = "No stored sessions";
430 }
431 return ResumeAction::Failed;
432 }
433 }
434
435 if (!store.load(id, state, error)) {
436 return ResumeAction::Failed;
437 }
438 if (state.status == "complete") {
439 log::progress("Session " + id + " is already complete");
440 out_ << state.result << "\n";
441 return ResumeAction::AlreadyComplete;
442 }
443 return ResumeAction::Continue;
444}

446int App::continueSession(const std::string& config_path, const config::Config& config,
447 const config::ModelConfig& selected, const std::shared_ptr<session::SessionStore>& store,
448 session::SessionState& state) {
449 std::string error;
450 const std::size_t repaired = session::repairDanglingToolCalls(state.messages);
451 if (repaired > 0) {
452 log::progress("Repaired " + std::to_string(repaired) + " unfinished tool call(s)");
453 }
454 state.status = "running";
455 state.error.clear();
456
457 Runtime runtime;
458 if (!buildRuntime(config_path, config, selected, runtime)) {
459 return 1;
460 }
461
462 // The stored conversation keeps its own system message. Replacing its text
463 // lets AGENTS.md and skill changes take effect on the resumed session.
464 const std::string systemPrompt = buildSystemPrompt(config.agent.system_prompt, runtime.fragments);
465 if (!systemPrompt.empty() && !state.messages.empty() && state.messages.front().role == "system") {
466 state.messages.front().content = systemPrompt;
467 }
468
469 // Persist the repaired conversation before the first model call, so another
470 // teardown during the resume does not repeat the repair from a stale file.
471 if (!store->save(state, error)) {
472 reportError(error);
473 return 1;
474 }
475
476 auto recorder = std::make_shared<session::SessionRecorder>(store, state);
477 agent::AgentLoopConfig loopConfig;
478 loopConfig.system_prompt = systemPrompt;
479 loopConfig.workspace = config.agent.workspace;
480 loopConfig.history = state.messages;
481 loopConfig.stop_requested = [] { return platform::shutdownRequested(); };
482 agent::AgentLoop loop(loopConfig, runtime.model, runtime.context, runtime.tools, recorder);
483
484 std::string result;
485 if (!loop.run("", result, error)) {
486 reportError(error);
487 return 1;
488 }
489 log::progress("Done");
490 out_ << result << "\n";
491 return 0;
492}

494int App::resumeTask(const cli::Invocation& invocation) {
495 std::string error;
496 std::string configPath = invocation.configPath;
497 if (configPath.empty()) {
498 configPath = config::findDefaultConfig(error);
499 if (configPath.empty()) {
500 reportError(error);
501 return 1;
502 }
503 }
504
505 config::Config config;
506 const config::ModelConfig* selected = nullptr;
507 if (!prepareModel(configPath, config, selected)) {
508 return 1;
509 }
510
511 std::string directory;
512 if (!sessionDirectoryFor(configPath, config, directory)) {
513 return 1;
514 }
515 const std::shared_ptr<session::SessionStore> store = openSessionStore(directory);
516 if (!store) {
517 return 1;
518 }
519
520 session::SessionState state;
521 const ResumeAction action = loadResumeSession(*store, invocation.session, state, error);
522 if (action == ResumeAction::Failed) {
523 reportError(error);
524 return 1;
525 }
526 if (action == ResumeAction::AlreadyComplete) {
527 return 0;
528 }
529
530 return continueSession(configPath, config, *selected, store, state);

573int App::runAcp(const cli::Invocation& invocation) {
574 std::string error;
575 const std::string& configPath = invocation.configPath;
576 config::Config config;
577 const config::ModelConfig* selected = nullptr;
578 if (!prepareModel(configPath, config, selected)) {
579 return 1;
580 }
581
582 // A session store is optional. Without one the agent serves the editor but
583 // cannot list, load or resume a conversation.
584 std::shared_ptr<session::SessionStore> store;
585 if (!config.agent.session_dir.empty()) {
586 std::string directory;
587 if (!sessionDirectoryFor(configPath, config, directory)) {
588 return 1;
589 }
590 store = openSessionStore(directory);
591 if (!store) {
592 return 1;
593 }
594 }
595
596 // The editor can pick the model of a session, so every configured model is
597 // offered by name.
598 std::vector<std::string> models;
599 std::map<std::string, config::ModelConfig> byName;
600 for (const config::ModelConfig& model : config.models) {
601 models.push_back(model.model);
602 byName[model.model] = model;
603 }
604
605 // Each session builds its own runtime, because the editor names the
606 // workspace, the model and the MCP servers it wants.
607 const auto factory = [this, &config, &configPath, &byName, selected](const acp::RuntimeRequest& request,
608 acp::SessionRuntime& runtime,
609 std::string& factoryError) {
610 const config::ModelConfig* model = selected;
611 if (!request.model.empty()) {
612 auto entry = byName.find(request.model);
613 if (entry != byName.end()) {
614 model = &entry->second;
615 }
616 }
617 // The servers the editor named win over the configured ones by name.
618 std::map<std::string, config::McpServerConfig> servers = config.mcp_servers;
619 for (const auto& [name, server] : request.mcp_servers) {
620 servers[name] = server;
621 }
622
623 RuntimeSetup setup;
624 setup.workspace = request.workspace;
625 setup.mcp_servers = servers;
626
627 Runtime built;
628 if (!buildRuntimeFor(configPath, config, *model, setup, built)) {
629 factoryError = "The session runtime could not be built";
630 return false;
631 }
632 runtime.model = built.model;
633 runtime.context = built.context;
634 runtime.tools = built.tools;
635 runtime.system_prompt = buildSystemPrompt(config.agent.system_prompt, built.fragments);
636 return true;
637 };
638
639 auto transport = options_.make_transport ? options_.make_transport() : std::make_shared<acp::StdioTransport>();
640 auto endpoint = std::make_shared<jsonrpc::Endpoint>(transport);
641 auto server = std::make_shared<acp::AcpServer>(endpoint, store, factory, models, selected->model);
642 endpoint->setRequestHandler([server](const std::string& method, const json::Value& params,
643 json::Value& result, jsonrpc::Error& error) {
644 return server->handle(method, params, result, error);
645 });
646 endpoint->setNotificationHandler([server](const std::string& method, const json::Value& params) {
647 server->notify(method, params);
648 });
649
650 log::progress("Serving the Agent Client Protocol");
651 endpoint->start();
652 // The connection ends when the editor closes it or when the process is
653 // asked to stop.
654 while (!endpoint->closed() && !platform::shutdownRequested()) {
655 std::this_thread::sleep_for(std::chrono::milliseconds(100));
656 }
657 log::progress("Connection closed");
658 return 0;
659}
