25bool splitAuthority(const std::string& authority, Url& url, std::string& error) {
26 std::string host = authority;
27 std::string port;
28
29 if (!host.empty() && host.front() == '[') {
30 const size_t close = host.find(']');
31 if (close == std::string::npos) {
32 error = "Endpoint has an unterminated IPv6 host: " + authority;
33 return false;
34 }
35 const std::string after = host.substr(close + 1);
36 if (!after.empty()) {
37 if (after.front() != ':') {
38 error = "Endpoint has an invalid host suffix: " + authority;
39 return false;
40 }
41 port = after.substr(1);
42 }
43 host = host.substr(0, close + 1);
44 } else {
45 const size_t colon = host.rfind(':');
46 if (colon != std::string::npos) {
47 port = host.substr(colon + 1);
48 host = host.substr(0, colon);
49 }
50 }
51
52 if (host.empty() || host == "[]") {
53 error = "Endpoint is missing a host: " + authority;
54 return false;
55 }
56 if (!port.empty()) {
57 if (!allDigits(port)) {
58 error = "Endpoint port is not numeric: " + port;
59 return false;
60 }

62 }

71bool parseUrl(const std::string& text, Url& out, std::string& error) {
72 const size_t schemeEnd = text.find("://");
