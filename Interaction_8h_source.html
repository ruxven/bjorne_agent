21// The seam between the agent and a human. In a container there is usually no
22// human and stdin already carried the task, so the default channel is
23// unavailable and fails fast instead of waiting for an answer.
24class Channel {
25 public:
26 virtual ~Channel() = default;
27
28 // False when ask can only fail. The tool reports the reason instead of
29 // blocking.
30 virtual bool available() const = 0;
31
32 // Asks the question and stores the answer. Returns false with a reason when
33 // no answer can be collected.
34 virtual bool ask(const Request& request, std::string& answer, std::string& error) = 0;
35};
36
37// The default channel. It never reads input; it reports why no human can be
38// reached.
39class UnavailableChannel final : public Channel {
40 public:
41 explicit UnavailableChannel(std::string reason);
42
43 bool available() const override { return false; }
44 bool ask(const Request& request, std::string& answer, std::string& error) override;
45
46 private:
47 std::string reason_;
48};
