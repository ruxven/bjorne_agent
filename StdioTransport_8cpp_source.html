44bool StdioTransport::sendLine(const std::string& line, std::string& error) {
45 // The Windows CRT treats a write to an invalid descriptor as a fatal error,
46 // so it is checked here rather than left to the write.
47 if (output_ < 0) {
48 error = "Cannot write to standard output";
49 return false;
50 }
51 const std::string payload = line + "\n";
52#if defined(_WIN32)
53 const int written = ::_write(output_, payload.data(), static_cast<unsigned int>(payload.size()));
54#else
55 const ssize_t written = ::write(output_, payload.data(), payload.size());
56#endif
57 if (written < 0 || static_cast<std::size_t>(written) != payload.size()) {
58 error = "Cannot write to standard output";
59 return false;
60 }
61 // The editor waits for the message before it sends the next one, so the
62 // write has to reach it before the call returns.
63 if (standard_output_ && std::fflush(stdout) != 0) {
64 error = "Cannot flush standard output";
65 return false;
66 }
67 return true;
68}

70bool StdioTransport::waitForInput(int timeout_ms) {
71 if (input_ < 0) {
72 return false;
73 }
74#if defined(_WIN32)
75 const HANDLE handle = reinterpret_cast<HANDLE>(static_cast<std::intptr_t>(_get_osfhandle(input_)));
76 if (handle == INVALID_HANDLE_VALUE || handle == nullptr) {
77 return false;
78 }
79 // WaitForSingleObject does not tell whether an anonymous pipe has data: the
80 // pipe handle reports itself ready, and the read that follows then blocks
81 // until the peer writes. PeekNamedPipe answers the question the deadline
82 // needs, so a partial line times out instead of wedging the reader. A console
83 // input handle is not a pipe and is waitable, so it keeps the wait.
84 if (::GetFileType(handle) != FILE_TYPE_PIPE) {
85 return ::WaitForSingleObject(handle, static_cast<DWORD>(timeout_ms)) == WAIT_OBJECT_0;
86 }
87 const ULONGLONG deadline = ::GetTickCount64() + static_cast<ULONGLONG>(timeout_ms);
88 for (;;) {
89 DWORD available = 0;
90 if (::PeekNamedPipe(handle, nullptr, 0, nullptr, &available, nullptr) == 0) {
91 // The pipe is gone. Let the read report the end or the error.
92 return true;
93 }
94 if (available > 0) {
95 return true;
96 }
97 if (::GetTickCount64() >= deadline) {
98 return false;
99 }
100 ::Sleep(5);
101 }
102#else
103 struct pollfd descriptor {};
104 descriptor.fd = input_;
105 descriptor.events = POLLIN;
106 const int status = ::poll(&descriptor, 1, timeout_ms);
107 if (status <= 0) {
108 return false;
109 }
110 // A hangup on the pipe means the editor closed the connection. Treating it
111 // as ready lets the read report the end of the stream.
112 return (descriptor.revents & (POLLIN | POLLHUP | POLLERR)) != 0;
113#endif
114}

116bool StdioTransport::readChunk(std::string& error) {
117 if (input_ < 0) {
118 error = "Cannot read from standard input";
119 return false;
120 }
121 char chunk[kChunkSize];
122#if defined(_WIN32)
123 const int read = ::_read(input_, chunk, static_cast<unsigned int>(sizeof(chunk)));
124#else
125 const ssize_t read = ::read(input_, chunk, sizeof(chunk));
126#endif
127 if (read < 0) {
128#if !defined(_WIN32)
129 if (errno == EINTR || errno == EAGAIN) {
130 return true;
131 }
132#endif
133 error = "Cannot read from standard input";
134 return false;
135 }
136 if (read == 0) {
137 error = "Standard input closed";
138 return false;
139 }
140 buffer_.append(chunk, static_cast<std::size_t>(read));
141 return true;
142}

144jsonrpc::ReadResult StdioTransport::readLine(int timeout_ms, std::string& line, std::string& error) {
145 const auto deadline = std::chrono::steady_clock::now() + std::chrono::milliseconds(timeout_ms);
146 for (;;) {
147 const std::size_t newline = buffer_.find('\n');
148 if (newline != std::string::npos) {

151 // A peer on either platform may end a line with a newline pair.

153 line.pop_back();

156 }
157

164 if (!waitForInput(static_cast<int>(remaining > 0 ? remaining : 1))) {

170 // peer that closes without a newline still means it.
