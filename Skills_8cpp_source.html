53// Reads one line without its terminator and moves position past it. Both LF
54// and CRLF are accepted, so a file written on Windows works unchanged.
55std::string nextLine(const std::string& text, std::size_t& position) {
56 const std::size_t start = position;
57 while (position < text.size() && text[position] != '\n') {
58 ++position;
59 }
60 std::string line = text.substr(start, position - start);
61 if (!line.empty() && line.back() == '\r') {
62 line.pop_back();
63 }
64 if (position < text.size()) {
65 ++position;
66 }
67 return line;
68}

70// A minimal key: value scanner. The frontmatter is a flat map, which is all
71// the skill format needs; a YAML library would breach the no dependency rule.
72bool parseFrontmatter(const std::string& text, std::map<std::string, std::string>& values, std::string& body,
73 std::string& reason) {
74 std::size_t position = 0;
75 // An editor may write a UTF-8 byte order mark before the opening delimiter.
76 if (text.rfind("\xEF\xBB\xBF", 0) == 0) {
77 position = 3;
78 }
79 if (trim(nextLine(text, position)) != "---") {
80 reason = "missing frontmatter";
81 return false;
82 }
83
84 bool closed = false;
85 while (position < text.size()) {
86 const std::string line = nextLine(text, position);
87 if (trim(line) == "---") {
88 closed = true;
89 break;
90 }
91 const std::size_t colon = line.find(':');
92 if (colon == std::string::npos) {
93 continue;
94 }
95 const std::string key = trim(line.substr(0, colon));
96 if (key.empty()) {
97 continue;
98 }
99 values[key] = unquote(trim(line.substr(colon + 1)));
100 }
101 if (!closed) {
102 reason = "unterminated frontmatter";
103 return false;
104 }
105
106 body = text.substr(position);
107 if (body.rfind("\r\n", 0) == 0) {
108 body.erase(0, 2);
109 } else if (!body.empty() && body.front() == '\n') {
110 body.erase(0, 1);
111 }
112 return true;
113}

115bool parseSkillFile(const std::string& path, Skill& skill, std::string& reason) {
116 std::string text;
117 if (!readFile(path, text)) {
118 reason = "cannot read file";
119 return false;
120 }
121
122 std::map<std::string, std::string> values;
123 std::string body;
124 if (!parseFrontmatter(text, values, body, reason)) {
125 return false;
126 }
127
128 const auto name = values.find("name");
129 if (name == values.end() || trim(name->second).empty()) {
130 reason = "missing name";
131 return false;
132 }
133 const auto description = values.find("description");
134 if (description == values.end() || trim(description->second).empty()) {
135 reason = "missing description";
136 return false;
137 }
138
139 skill.name = trim(name->second);
140 skill.description = trim(description->second);
141 const std::size_t first = body.find_first_not_of("\r\n");
142 if (first == std::string::npos) {
143 skill.instructions.clear();
144 } else {
145 const std::size_t last = body.find_last_not_of(" \t\r\n");
146 skill.instructions = body.substr(first, last - first + 1);
147 }
148 skill.path = path;
149 return true;
150}

154SkillLoad loadSkills(const std::string& directory) {
155 SkillLoad result;
156 if (directory.empty()) {
157 return result;
158 }
159
160 std::error_code code;
161 if (!std::filesystem::is_directory(directory, code) || code) {
162 return result;
163 }
164 result.directory_found = true;
165
166 std::vector<std::filesystem::path> skillDirectories;
167 std::filesystem::directory_iterator entry(directory, code);
168 const std::filesystem::directory_iterator end;
169 while (!code && entry != end) {
170 std::error_code entryCode;
171 if (entry->is_directory(entryCode) && !entryCode) {
172 skillDirectories.push_back(entry->path());
173 }
174 entry.increment(code);
175 }
176 std::sort(skillDirectories.begin(), skillDirectories.end());
177
178 for (const std::filesystem::path& skillDirectory : skillDirectories) {
179 const std::filesystem::path file = skillDirectory / kSkillFileName;
180 std::error_code fileCode;
181 if (!std::filesystem::is_regular_file(file, fileCode) || fileCode) {
182 continue;
183 }
184
185 Skill skill;
186 std::string reason;
187 if (parseSkillFile(file.string(), skill, reason)) {
188 result.skills.push_back(std::move(skill));
189 } else {
190 result.skipped.push_back(file.string() + ": " + reason);
191 }
192 }
193 return result;
194}
