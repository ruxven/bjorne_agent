51// True for a path that starts at the root: /dir on POSIX, C:\dir or \\server
52// on Windows.
53bool isAbsolutePath(const std::string& path) {
54 if (path.empty()) {
55 return false;
56 }
57 if (isPathSeparator(path.front())) {
58 return true;
59 }
60#if defined(_WIN32)
61 if (path.size() >= 3 && std::isalpha(static_cast<unsigned char>(path[0])) != 0 && path[1] == ':' &&
62 isPathSeparator(path[2])) {
63 return true;
64 }
65#endif
66 return false;
67}

77std::string resolveAgainstConfig(const std::string& configPath, const std::string& path) {
78 if (path.empty() || isAbsolutePath(path)) {
79 return path;
80 }
81 const size_t slash = lastPathSeparator(configPath);
82 if (slash == std::string::npos) {
83 return path;
84 }
85 return configPath.substr(0, slash + 1) + path;
86}

113// Fills in the context configuration and loads the compression prompt. The
114// prompt path is relative to the config file, so it is resolved against the
115// config path rather than the working directory.
116bool buildContextConfig(const std::string& configPath, const bjorne_agent::config::ModelConfig& model,
117 bjorne_agent::context::ContextManagerConfig& contextConfig) {
118 contextConfig.context_window = model.context_window;
119 contextConfig.max_tokens = model.max_tokens;
120 if (model.compression_prompt_path.empty()) {
121 return true;
122 }
123
124 std::string prompt;
125 const std::string promptPath = resolveAgainstConfig(configPath, model.compression_prompt_path);
126 bjorne_agent::log::detail("Compression prompt: " + promptPath);

129 bjorne_agent::log::fail(message);
130 std::cerr << message << "\n";
131 return false;
132 }
133 contextConfig.compression_prompt = prompt;
134 return true;
135}

158// Runs one task: loads the config, builds the model, context and tools, and
159// prints the final answer. Returns 0 on success and 1 after reporting an error.
160int runTask(const bjorne_agent::cli::Invocation& invocation) {
161 std::string error;

163 bjorne_agent::log::progress("Loading config: " + configPath);
164

166 if (!bjorne_agent::config::load(configPath, config, error)) {
167 bjorne_agent::log::fail(error);
168 std::cerr << error << "\n";

170 }
171
172 const bjorne_agent::config::ModelConfig* selected = selectModel(config.models);
173 if (selected == nullptr) {
174 bjorne_agent::log::fail("Config has no models");
175 std::cerr << "Config has no models\n";
176 return 1;
177 }
178 bjorne_agent::log::progress("Selected model: " + selected->model + " at " + selected->endpoint);
179 bjorne_agent::log::detail("API key comes from environment variable: " + selected->api_key_env);
180 bjorne_agent::log::detail("Workspace: " + config.agent.workspace);
181 if (selected->ignore_self_signed) {
182 bjorne_agent::log::progress("TLS peer verification is off (ignore_self_signed is set)");
183 }
184 if (!selected->hostname.empty()) {

186 }

188 // The task comes from the command line when one was given, otherwise from
189 // standard input.

193 return 1;

196
197 bjorne_agent::context::ContextManagerConfig contextConfig;
198 if (!buildContextConfig(configPath, *selected, contextConfig)) {
199 return 1;
200 }
201
202 bjorne_agent::model::ModelClientOptions modelOptions;
203 modelOptions.endpoint = selected->endpoint;
204 modelOptions.model = selected->model;
205 modelOptions.api_key_env = selected->api_key_env;
206 modelOptions.ignore_self_signed = selected->ignore_self_signed;
207 modelOptions.hostname = selected->hostname;
208 auto model = std::make_shared<bjorne_agent::model::ModelClient>(std::move(modelOptions));
209 model->setTools(bjorne_agent::tools::ToolRegistry::definitionsJson());
210
211 auto context = std::make_shared<bjorne_agent::context::ContextManager>(contextConfig, makeCompressionHandler(model));
212 auto tools = std::make_shared<bjorne_agent::tools::ToolRegistry>(config.agent.workspace);
213
214 bjorne_agent::agent::AgentLoopConfig loopConfig;
215 loopConfig.system_prompt = config.agent.system_prompt;
216 loopConfig.workspace = config.agent.workspace;
217 bjorne_agent::agent::AgentLoop loop(loopConfig, model, context, tools);
218
219 std::string result;
220 if (!loop.run(task, result, error)) {
221 bjorne_agent::log::fail(error);
222 std::cerr << error << "\n";
223 return 1;
224 }
225 bjorne_agent::log::progress("Done");
226 std::cout << result << "\n";
