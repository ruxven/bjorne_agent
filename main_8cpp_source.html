51#if defined(_WIN32)
52 return c == '/' || c == '\\';
53#else
54 return c == '/';

58// True for a path that starts at the root: /dir on POSIX, C:\dir or \\server
59// on Windows.
60bool isAbsolutePath(const std::string& path) {
61 if (path.empty()) {
62 return false;
63 }
64 if (isPathSeparator(path.front())) {
65 return true;
66 }
67#if defined(_WIN32)
68 if (path.size() >= 3 && std::isalpha(static_cast<unsigned char>(path[0])) != 0 && path[1] == ':' &&
69 isPathSeparator(path[2])) {
70 return true;
71 }
72#endif
73 return false;
74}

76size_t lastPathSeparator(const std::string& path) {
77#if defined(_WIN32)

79#else
80 return path.find_last_of('/');
81#endif

84std::string resolveAgainstConfig(const std::string& configPath, const std::string& path) {
85 if (path.empty() || isAbsolutePath(path)) {
86 return path;
87 }
88 const size_t slash = lastPathSeparator(configPath);
89 if (slash == std::string::npos) {
90 return path;
91 }
92 return configPath.substr(0, slash + 1) + path;
93}

125// Fills in the context configuration and loads the compression prompt. The
126// prompt path is relative to the config file, so it is resolved against the
127// config path rather than the working directory.
128bool buildContextConfig(const std::string& configPath, const bjorne_agent::config::ModelConfig& model,
129 bjorne_agent::context::ContextManagerConfig& contextConfig) {
130 contextConfig.context_window = model.context_window;
131 contextConfig.max_tokens = model.max_tokens;
132 if (model.compression_prompt_path.empty()) {
133 return true;
134 }
135
136 std::string prompt;
137 const std::string promptPath = resolveAgainstConfig(configPath, model.compression_prompt_path);
138 bjorne_agent::log::detail("Compression prompt: " + promptPath);

141 return false;

147// Builds the callback that asks the model to summarize a conversation when
148// the context no longer fits.
149bjorne_agent::context::CompressionHandler makeCompressionHandler(const std::shared_ptr<bjorne_agent::model::ModelClient>& model) {
150 return [model](const std::vector<bjorne_agent::model::ModelMessage>& messages, const std::string& instruction,
151 std::string& summary, std::string& handlerError) {
152 std::vector<bjorne_agent::model::ModelMessage> request;
153 bjorne_agent::model::ModelMessage system;
154 system.role = "system";
155 system.content = instruction;

168// Builds the system prompt from the configured text and the assembled
169// fragments. Skills come first and the nearest AGENTS.md last, so the most
170// specific instruction is the last thing the model reads.
171std::string buildSystemPrompt(const std::string& base, const std::vector<bjorne_agent::context::Fragment>& fragments) {
172 std::string prompt = base;
173 const std::string assembled = bjorne_agent::context::renderFragments(fragments);
174 if (assembled.empty()) {
175 return prompt;
176 }
177 if (!prompt.empty()) {
178 prompt += "\n\n";
179 }
180 prompt += assembled;
181 return prompt;

223bool buildRuntime(const std::string& configPath, const bjorne_agent::config::Config& config,
224 const bjorne_agent::config::ModelConfig& selected, Runtime& out) {
225 if (!buildContextConfig(configPath, selected, out.context_config)) {
226 return false;
227 }
228
229 // Skills and AGENTS.md are assembled before the model client, because the
230 // system prompt they produce is part of the request and load_skill has to be
231 // in the tool set from the first turn.
232 const std::string skillsDirectory = resolveAgainstConfig(configPath, config.agent.skills_path);
233 const bjorne_agent::skills::SkillLoad skillLoad = bjorne_agent::skills::loadSkills(skillsDirectory);
234 if (!skillsDirectory.empty() && !skillLoad.directory_found) {
235 bjorne_agent::log::fail("Skills directory not found: " + skillsDirectory);
236 }
237 for (const std::string& skipped : skillLoad.skipped) {
238 bjorne_agent::log::fail("Skipping skill " + skipped);
239 }
240 for (const bjorne_agent::skills::Skill& skill : skillLoad.skills) {
241 bjorne_agent::log::detail("Skill: " + skill.name);
242 }
243
244 const bjorne_agent::context::AgentsFiles agents =
245 bjorne_agent::context::loadAgentsFiles(config.agent.workspace, bjorne_agent::config::userConfigDirectory());
246 for (const std::string& path : agents.unreadable) {
247 bjorne_agent::log::fail("Cannot read " + path);
248 }
249 for (const bjorne_agent::context::Fragment& fragment : agents.fragments) {
250 bjorne_agent::log::detail("Context: " + fragment.title);
251 }
252
253 if (!skillLoad.skills.empty()) {
254 out.fragments.push_back(bjorne_agent::skills::skillIndexFragment(skillLoad.skills));
255 }
256 out.fragments.insert(out.fragments.end(), agents.fragments.begin(), agents.fragments.end());
257
258 bjorne_agent::model::ModelClientOptions modelOptions;
259 modelOptions.endpoint = selected.endpoint;
260 modelOptions.model = selected.model;
261 modelOptions.api_key_env = selected.api_key_env;
262 modelOptions.ignore_self_signed = selected.ignore_self_signed;
263 modelOptions.hostname = selected.hostname;
264 out.model = std::make_shared<bjorne_agent::model::ModelClient>(std::move(modelOptions));
265
266 out.context =
267 std::make_shared<bjorne_agent::context::ContextManager>(out.context_config, makeCompressionHandler(out.model));
268 out.tools = std::make_shared<bjorne_agent::tools::ToolRegistry>(config.agent.workspace);
269 bjorne_agent::interaction::registerRequestInputTool(
270 *out.tools, bjorne_agent::interaction::makeChannel(config.agent.interactive));
271 if (!skillLoad.skills.empty()) {
272 auto skillCatalog = std::make_shared<std::vector<bjorne_agent::skills::Skill>>(skillLoad.skills);
273 bjorne_agent::skills::registerLoadSkillTool(*out.tools, skillCatalog);
274 }
275 out.model->setTools(out.tools->definitionsJson());
276 return true;
277}

306// Runs one task: loads the config, builds the model, context and tools, and
307// prints the final answer. Returns 0 on success and 1 after reporting an error.
308int runTask(const bjorne_agent::cli::Invocation& invocation) {
309 std::string error;
310 const std::string& configPath = invocation.configPath;
311 bjorne_agent::config::Config config;
312 const bjorne_agent::config::ModelConfig* selected = nullptr;
313 if (!prepareModel(configPath, config, selected)) {
314 return 1;
315 }
316
317 // The task comes from the command line when one was given, otherwise from
318 // standard input.
319 std::string task = invocation.taskFromArgument ? invocation.task : readStdin();
320 if (task.empty()) {
321 std::cerr << bjorne_agent::cli::usage();
322 return 1;
323 }
324 bjorne_agent::log::progress("Task: " + task);
325
326 Runtime runtime;
327 if (!buildRuntime(configPath, config, *selected, runtime)) {
328 return 1;
329 }
330
331 // With a session directory the run creates a record before the first model
332 // call and rewrites it after each turn. Without one the run stays one-shot.
333 std::shared_ptr<bjorne_agent::session::SessionStore> store;
334 std::shared_ptr<bjorne_agent::session::SessionRecorder> recorder;
335 if (!config.agent.session_dir.empty()) {
336 std::string directory;
337 if (!sessionDirectoryFor(configPath, config, directory)) {
338 return 1;
339 }
340 store = openSessionStore(directory);
341 if (!store) {
342 return 1;
343 }
344
345 bjorne_agent::session::SessionState state;
346 state.task = task;
347 state.workspace = config.agent.workspace;
348 state.model = selected->model;
349 if (!store->begin(state, error)) {
350 reportError(error);
351 return 1;
352 }
353 bjorne_agent::log::progress("Session: " + state.id);
354 recorder = std::make_shared<bjorne_agent::session::SessionRecorder>(store, std::move(state));
355 }
356
357 bjorne_agent::agent::AgentLoopConfig loopConfig;
358 loopConfig.system_prompt = buildSystemPrompt(config.agent.system_prompt, runtime.fragments);
359 loopConfig.workspace = config.agent.workspace;
360 loopConfig.stop_requested = [] { return bjorne_agent::platform::shutdownRequested(); };
361 bjorne_agent::agent::AgentLoop loop(loopConfig, runtime.model, runtime.context, runtime.tools, recorder);
362
363 std::string result;
364 if (!loop.run(task, result, error)) {
365 reportError(error);
366 return 1;
367 }
368 bjorne_agent::log::progress("Done");
369 std::cout << result << "\n";
370 return 0;
371}

373// The outcome of looking up the session a resume asked for. A completed
374// session is not a failure: its stored answer is printed and no model runs.
375enum class ResumeAction { Continue, AlreadyComplete, Failed };
376
377ResumeAction loadResumeSession(const bjorne_agent::session::SessionStore& store, const std::string& requested,
378 bjorne_agent::session::SessionState& state, std::string& error) {
379 std::string id = requested;
380 if (id.empty()) {
381 id = store.latest(error);
382 if (id.empty()) {
383 if (error.empty()) {
384 error = "No stored sessions";
385 }
386 return ResumeAction::Failed;
387 }
388 }
389
390 if (!store.load(id, state, error)) {
391 return ResumeAction::Failed;
392 }
393 if (state.status == "complete") {
394 bjorne_agent::log::progress("Session " + id + " is already complete");
395 std::cout << state.result << "\n";
396 return ResumeAction::AlreadyComplete;
397 }
398 return ResumeAction::Continue;
399}

401// Continues a stored session. The stored system message is refreshed from the
402// current config, and an unfinished tool call gets a synthetic result so the
403// sequence is valid again.
404int resumeTask(const bjorne_agent::cli::Invocation& invocation) {
405 std::string error;
406 std::string configPath = invocation.configPath;
407 if (configPath.empty()) {
408 configPath = bjorne_agent::config::findDefaultConfig(error);
409 if (configPath.empty()) {
410 reportError(error);
411 return 1;
412 }
413 }
414
415 bjorne_agent::config::Config config;
416 const bjorne_agent::config::ModelConfig* selected = nullptr;
417 if (!prepareModel(configPath, config, selected)) {
418 return 1;
419 }
420
421 std::string directory;
422 if (!sessionDirectoryFor(configPath, config, directory)) {
423 return 1;
424 }
425 const std::shared_ptr<bjorne_agent::session::SessionStore> store = openSessionStore(directory);
426 if (!store) {
427 return 1;
428 }
429
430 bjorne_agent::session::SessionState state;
431 const ResumeAction action = loadResumeSession(*store, invocation.session, state, error);
432 if (action == ResumeAction::Failed) {
433 reportError(error);
434 return 1;
435 }
436 if (action == ResumeAction::AlreadyComplete) {
437 return 0;
438 }
439
440 const std::size_t repaired = bjorne_agent::session::repairDanglingToolCalls(state.messages);
441 if (repaired > 0) {
442 bjorne_agent::log::progress("Repaired " + std::to_string(repaired) + " unfinished tool call(s)");
443 }
444 state.status = "running";
445 state.error.clear();
446
447 Runtime runtime;
448 if (!buildRuntime(configPath, config, *selected, runtime)) {
449 return 1;
450 }
451
452 // The stored conversation keeps its own system message. Replacing its text
453 // lets AGENTS.md and skill changes take effect on the resumed session.
454 const std::string systemPrompt = buildSystemPrompt(config.agent.system_prompt, runtime.fragments);
455 if (!systemPrompt.empty() && !state.messages.empty() && state.messages.front().role == "system") {
456 state.messages.front().content = systemPrompt;
457 }
458
459 // Persist the repaired conversation before the first model call, so another
460 // teardown during the resume does not repeat the repair from a stale file.
461 if (!store->save(state, error)) {
462 reportError(error);
463 return 1;
464 }
465
466 auto recorder = std::make_shared<bjorne_agent::session::SessionRecorder>(store, state);
467 bjorne_agent::agent::AgentLoopConfig loopConfig;
468 loopConfig.system_prompt = systemPrompt;
469 loopConfig.workspace = config.agent.workspace;
470 loopConfig.history = state.messages;
471 loopConfig.stop_requested = [] { return bjorne_agent::platform::shutdownRequested(); };
472 bjorne_agent::agent::AgentLoop loop(loopConfig, runtime.model, runtime.context, runtime.tools, recorder);
473
474 std::string result;
475 if (!loop.run("", result, error)) {
476 reportError(error);
477 return 1;
478 }
479 bjorne_agent::log::progress("Done");
480 std::cout << result << "\n";
481 return 0;
482}
