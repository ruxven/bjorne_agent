40bool parsePort(const std::string& port, unsigned short& value) {
41 if (port.empty()) {
42 return false;
43 }
44 unsigned long parsed = 0;
45 for (const char c : port) {
46 if (c < '0' || c > '9') {
47 return false;
48 }
49 parsed = parsed * 10 + static_cast<unsigned long>(c - '0');
50 if (parsed > 65535) {
51 return false;
52 }
53 }
54 if (parsed == 0) {
55 return false;
56 }
57 value = static_cast<unsigned short>(parsed);
58 return true;
59}

70bool numericAddress(const std::string& host, unsigned short port, ResolvedAddress& out) {
71 const std::string bare = withoutBrackets(host);
72 sockaddr_in v4{};
73 if (platform::addressToBinary(AF_INET, bare.c_str(), &v4.sin_addr) == 1) {
74 v4.sin_family = AF_INET;
75 v4.sin_port = htons(port);

93bool readFile(const std::string& path, std::string& contents) {
94 const int fd = ::open(path.c_str(), O_RDONLY);
95 if (fd < 0) {
96 return false;
97 }
98 char buffer[4096];
99 for (;;) {
100 const ssize_t bytes = ::read(fd, buffer, sizeof(buffer));
101 if (bytes < 0) {
102 ::close(fd);
103 return false;
104 }
105 if (bytes == 0) {
106 break;
107 }
108 contents.append(buffer, static_cast<size_t>(bytes));
109 }
110 ::close(fd);
111 return true;
112}

114std::vector<std::string> splitFields(const std::string& line) {
115 std::vector<std::string> fields;
116 size_t position = 0;
117 while (position < line.size()) {
118 while (position < line.size() && std::isspace(static_cast<unsigned char>(line[position])) != 0) {
119 ++position;
120 }
121 const size_t start = position;
122 while (position < line.size() && std::isspace(static_cast<unsigned char>(line[position])) == 0) {
123 ++position;
124 }
125 if (start != position) {
126 fields.push_back(line.substr(start, position - start));
127 }
128 }
129 return fields;
130}

132// One /etc/hosts line: an address, then one or more names for it.
133bool hostsLineMatches(const std::string& line, const std::string& host, std::string& address) {
134 const size_t comment = line.find('#');
135 const std::vector<std::string> fields = splitFields(comment == std::string::npos ? line : line.substr(0, comment));
136 if (fields.size() < 2) {
137 return false;
138 }
139 for (size_t i = 1; i < fields.size(); ++i) {
140 if (::strcasecmp(fields[i].c_str(), host.c_str()) == 0) {
141 address = fields[0];
142 return true;
143 }
144 }
145 return false;
146}

148bool lookupHosts(const std::string& path, const std::string& host, unsigned short port,
149 std::vector<ResolvedAddress>& addresses) {
150 std::string contents;
151 if (!readFile(path, contents)) {
152 return false;
153 }
154 size_t start = 0;
155 for (;;) {
156 const size_t end = contents.find('\n', start);
157 const std::string line = contents.substr(start, end == std::string::npos ? std::string::npos : end - start);
158 std::string address;
159 ResolvedAddress resolved;
160 if (hostsLineMatches(line, host, address) && numericAddress(address, port, resolved)) {
161 addresses.push_back(resolved);
162 }
163 if (end == std::string::npos) {
164 break;
165 }
166 start = end + 1;

201bool resolveWithGetaddrinfo(const std::string& host, const std::string& port,
202 std::vector<ResolvedAddress>& addresses, std::string& error) {
203 addrinfo hints{};
204 hints.ai_family = AF_UNSPEC;
205 hints.ai_socktype = SOCK_STREAM;
206 addrinfo* results = nullptr;
207 const int rc = ::getaddrinfo(host.c_str(), port.c_str(), &hints, &results);
208 if (rc != 0) {
209 error = std::string("DNS resolution failed: ") + platform::addressErrorMessage(rc);
210 return false;
211 }
212 for (addrinfo* entry = results; entry != nullptr; entry = entry->ai_next) {
213 if (entry->ai_addrlen > sizeof(ResolvedAddress::storage)) {
214 continue;
215 }
216 ResolvedAddress address;
217 std::memcpy(&address.storage, entry->ai_addr, entry->ai_addrlen);
218 address.length = static_cast<platform::SocketLength>(entry->ai_addrlen);
219 addresses.push_back(address);
220 }
221 ::freeaddrinfo(results);
222 if (addresses.empty()) {
223 error = "DNS resolution failed: no addresses";
224 return false;
225 }
226 return true;
227}

233bool resolveHost(const std::string& host, const std::string& port, std::vector<ResolvedAddress>& addresses,
234 std::string& error, const ResolverOptions& options) {
235 addresses.clear();
236 if (!platform::initializeSockets()) {
