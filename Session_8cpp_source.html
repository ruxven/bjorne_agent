172bool fromJson(const std::string& text, SessionState& state, std::string& error) {
173 json::Value root;
174 try {
175 root = json::parse(text);
176 } catch (const std::exception& exception) {
177 error = std::string("Cannot parse session file: ") + exception.what();
178 return false;
179 }
180 if (!root.isObject()) {
181 error = "Session file is not a JSON object";
182 return false;
183 }
184
185 long long version = kSessionVersion;
186 getNumber(root, "version", version);
187 if (version != kSessionVersion) {
188 error = "Unsupported session version: " + std::to_string(version);
189 return false;
190 }
191
192 getString(root, "id", state.id);
193 getNumber(root, "created", state.created);
194 getNumber(root, "updated", state.updated);
195 getString(root, "task", state.task);
196 getString(root, "workspace", state.workspace);
197 getString(root, "model", state.model);
198 getString(root, "status", state.status);
199 getString(root, "result", state.result);
200 getString(root, "error", state.error);
201 getNumber(root, "turns", state.turns);
202 if (state.status.empty()) {
203 state.status = "unknown";
204 }
205
206 const json::Value* messages = root.find("messages");
207 if (messages != nullptr) {
208 if (!messages->isArray()) {
209 error = "Session messages must be an array";
210 return false;
211 }
212 for (const json::Value& node : messages->array) {
213 model::ModelMessage message;
214 if (!parseMessage(node, message)) {
215 error = "Malformed message in session file";
216 return false;
217 }
218 state.messages.push_back(std::move(message));
219 }
220 }
221 return true;
222}

225 const std::string temporary = path + kTemporarySuffix;

228 if (!file.is_open()) {

232 file << text;

244 std::error_code code;
245 std::filesystem::remove(temporary, code);
246 error = "Cannot replace session file: " + path;
247 return false;
248 }

267bool isValidSessionId(const std::string& id) {
268 if (id.empty() || id.size() > 128 || id == "." || id == "..") {
269 return false;
270 }
271 for (const char character : id) {
272 const unsigned char value = static_cast<unsigned char>(character);
273 if (std::isalnum(value) == 0 && character != '-' && character != '_' && character != '.') {
274 return false;
275 }
276 }
277 return true;

386 std::vector<SessionSummary> summaries;
387 error.clear();

389 std::error_code code;
390 std::filesystem::directory_iterator entry(directory_, code);
391 const std::filesystem::directory_iterator end;

396 std::string text;

405 summary.status = state.status;
406 summary.task = oneLineTask(state.task);
407 summary.messages = state.messages.size();

409 }

413 }
414

417 return {};
418 }
419
420 std::sort(summaries.begin(), summaries.end(), [](const SessionSummary& left, const SessionSummary& right) {
421 if (left.updated != right.updated) {
422 return left.updated > right.updated;
423 }
424 return left.id < right.id;
425 });
426 return summaries;

437std::size_t repairDanglingToolCalls(std::vector<model::ModelMessage>& messages) {
438 std::size_t repaired = 0;
439 for (std::size_t index = 0; index < messages.size(); ++index) {
440 if (messages[index].role != "assistant" || messages[index].tool_calls.empty()) {
441 continue;
442 }
443
444 // Tool results for an assistant message sit immediately after it. Stop at
445 // the first message of another kind, which is also where a missing result
446 // has to be inserted to keep the sequence valid.
447 std::size_t end = index + 1;
448 std::vector<std::string> answered;
449 while (end < messages.size() && messages[end].role == "tool") {
450 answered.push_back(messages[end].tool_call_id);
451 ++end;
452 }
453
454 std::vector<model::ModelMessage> missing;
455 for (const model::ToolCall& call : messages[index].tool_calls) {
456 if (std::find(answered.begin(), answered.end(), call.id) != answered.end()) {
457 continue;
458 }
459 model::ModelMessage tool;
460 tool.role = "tool";
461 tool.tool_call_id = call.id;
462 tool.content = "Error: the tool call did not finish before the session stopped.";
463 missing.push_back(std::move(tool));
464 ++repaired;
465 }
466
467 if (!missing.empty()) {
468 messages.insert(messages.begin() + static_cast<std::ptrdiff_t>(end), missing.begin(), missing.end());
469 }
470 index = end + missing.size() - 1;
471 }
472 return repaired;
473}
