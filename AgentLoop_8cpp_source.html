127bool AgentLoop::run(const std::string& task, std::string& result, std::string& error) {
128 stop_reason_.clear();
129 std::vector<model::ModelMessage> messages = buildContext(task);
130 if (messages.empty()) {
131 error = "No task and no session history to continue";
132 log::fail(error);
133 notifyFailure(messages, error);
134 return false;
135 }
136
137 for (long long iteration = 0; iteration < config_.max_iterations; ++iteration) {
138 if (stopRequested()) {
139 stop_reason_ = "cancelled";
140 return interrupt(messages, error);
141 }
142
143 model::ModelResponse response;
144 if (!callModel(messages, iteration, response, error)) {
145 return false;
146 }
147 // A shutdown that arrives while the request is in flight is noticed here,
148 // before the response is added, so a partial turn is not recorded.
149 if (stopRequested()) {
150 stop_reason_ = "cancelled";
151 return interrupt(messages, error);
152 }
153 narrate(response);
154
155 if (response.tool_calls.empty()) {
156 log::progress("Model returned a final response");
157 stop_reason_ = "end_turn";
158 model::ModelMessage assistant;
159 assistant.role = "assistant";
160 assistant.content = response.content;
161 messages.push_back(std::move(assistant));
162 result = response.content;
163 if (observer_) {
164 observer_->onComplete(messages, result);
165 }
166 return true;
167 }
168
169 // A permission gate answers before a tool runs. A denied call ends the
170 // turn as a refusal; a stop request that arrived while the question was in
171 // flight takes precedence and interrupts instead.
172 std::string denial;
173 if (!approved(response, denial)) {
174 if (stopRequested()) {
175 stop_reason_ = "cancelled";
176 return interrupt(messages, error);
177 }
178 stop_reason_ = "refusal";
179 result = denial;
180 if (observer_) {
181 observer_->onComplete(messages, result);
182 }
183 return true;
184 }
185
186 log::progress("Model requested " + std::to_string(response.tool_calls.size()) + " tool call(s)");
187 appendToolTurn(response, messages);
188 if (observer_) {
189 observer_->onTurn(messages, iteration + 1);
190 }
191 }
192 stop_reason_ = "max_turn_requests";
193 error = "Iteration cap reached without a final response";
194 log::fail(error);
195 notifyFailure(messages, error);
196 return false;
197}
