31void AgentLoop::appendToolTurn(const model::ModelResponse& response, std::vector<model::ModelMessage>& messages) const {
32 model::ModelMessage assistant;
33 assistant.role = "assistant";
34 assistant.content = response.content;
35 assistant.tool_calls = response.tool_calls;
36 messages.push_back(assistant);
37
38 for (const model::ToolCall& call : response.tool_calls) {

40 model::ModelMessage tool;
41 tool.role = "tool";
42 tool.tool_call_id = call.id;
43 tool.content = outcome.ok ? outcome.output : ("Error: " + outcome.output);

48bool AgentLoop::run(const std::string& task, std::string& result, std::string& error) {
49 std::vector<model::ModelMessage> messages = buildContext(task);
50 for (long long iteration = 0; iteration < config_.max_iterations; ++iteration) {
51 std::vector<model::ModelMessage> trimmed;
52 if (!context_->trim(messages, trimmed, error)) {
53 log::fail(error);
54 return false;
55 }
56 log::progress("Iteration " + std::to_string(iteration + 1) + " of " + std::to_string(config_.max_iterations) +
57 ", sending " + std::to_string(trimmed.size()) + " messages");
58 model::ModelResponse response;
59 if (!model_->respond(trimmed, response, error)) {
60 log::fail(error);
61 return false;
62 }
63 // Show what the model is thinking before anything it asked for runs. The
64 // final answer goes to stdout, so its content is not repeated here; a
65 // reasoning trace is always shown, including on the final turn.
66 if (!response.reasoning.empty()) {
67 log::block("thinking", response.reasoning);
68 }
69 if (!response.tool_calls.empty() && !response.content.empty()) {
70 log::block("assistant", response.content);
71 }
72 if (response.tool_calls.empty()) {
73 log::progress("Model returned a final response");
74 result = response.content;
75 return true;
76 }
77 log::progress("Model requested " + std::to_string(response.tool_calls.size()) + " tool call(s)");
78 appendToolTurn(response, messages);
79 }
80 error = "Iteration cap reached without a final response";
81 log::fail(error);
82 return false;
83}
