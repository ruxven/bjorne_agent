40void AgentLoop::appendToolTurn(const model::ModelResponse& response, std::vector<model::ModelMessage>& messages) const {
41 model::ModelMessage assistant;
42 assistant.role = "assistant";
43 assistant.content = response.content;
44 assistant.tool_calls = response.tool_calls;
45 messages.push_back(assistant);

47 for (const model::ToolCall& call : response.tool_calls) {

72bool AgentLoop::run(const std::string& task, std::string& result, std::string& error) {
73 std::vector<model::ModelMessage> messages = buildContext(task);
74 if (messages.empty()) {
75 error = "No task and no session history to continue";
76 log::fail(error);
77 notifyFailure(messages, error);
78 return false;
79 }
80
81 for (long long iteration = 0; iteration < config_.max_iterations; ++iteration) {
82 if (config_.stop_requested && config_.stop_requested()) {
83 return interrupt(messages, error);
84 }
85
86 std::vector<model::ModelMessage> trimmed;
87 if (!context_->trim(messages, trimmed, error)) {
88 log::fail(error);
89 notifyFailure(messages, error);
90 return false;
91 }
92 log::progress("Iteration " + std::to_string(iteration + 1) + " of " + std::to_string(config_.max_iterations) +
93 ", sending " + std::to_string(trimmed.size()) + " messages");
94 model::ModelResponse response;
95 if (!model_->respond(trimmed, response, error)) {
96 log::fail(error);
97 notifyFailure(messages, error);
98 return false;
99 }
100 // A shutdown that arrives while the request is in flight is noticed here,
101 // before the response is added, so a partial turn is not recorded.
102 if (config_.stop_requested && config_.stop_requested()) {
103 return interrupt(messages, error);
104 }
105 // Show what the model is thinking before anything it asked for runs. The
106 // final answer goes to stdout, so its content is not repeated here; a
107 // reasoning trace is always shown, including on the final turn.
108 if (!response.reasoning.empty()) {
109 log::block("thinking", response.reasoning);
110 }
111 if (!response.tool_calls.empty() && !response.content.empty()) {
112 log::block("assistant", response.content);
113 }
114 if (response.tool_calls.empty()) {
115 log::progress("Model returned a final response");
116 model::ModelMessage assistant;
117 assistant.role = "assistant";
118 assistant.content = response.content;
119 messages.push_back(std::move(assistant));
120 result = response.content;
121 if (observer_) {
122 observer_->onComplete(messages, result);
123 }
124 return true;
125 }
126 log::progress("Model requested " + std::to_string(response.tool_calls.size()) + " tool call(s)");
127 appendToolTurn(response, messages);
128 if (observer_) {
129 observer_->onTurn(messages, iteration + 1);
130 }
131 }
132 error = "Iteration cap reached without a final response";
133 log::fail(error);
134 notifyFailure(messages, error);
135 return false;
136}
