34// The options in the shape the protocol carries them in. The label is looked
35// up rather than indexed, so the two lists cannot drift apart.
36json::Value permissionOptions(const interaction::Request& request) {
37 json::Value options = arrayValue();
38 for (const std::string& label : request.options) {
39 for (const PermissionChoice& choice : permissionChoices()) {
40 if (choice.label != label) {
41 continue;
42 }
43 json::Value option = objectValue();
44 option.object["optionId"] = stringValue(choice.id);
45 option.object["name"] = stringValue(choice.label);
46 option.object["kind"] = stringValue(choice.kind);
47 options.array.push_back(std::move(option));
48 break;
49 }
50 }
51 return options;
52}

91bool AcpChannel::ask(const interaction::Request& request, std::string& answer, std::string& error) {
92 json::Value toolCall = objectValue();
93 toolCall.object["toolCallId"] = stringValue(tool_call_id_);
94 toolCall.object["title"] = stringValue(request.question);
95
96 json::Value params = objectValue();
97 params.object["sessionId"] = stringValue(session_id_);
98 params.object["toolCall"] = std::move(toolCall);
99 params.object["options"] = permissionOptions(request);
100
101 json::Value result;
102 jsonrpc::RequestOptions options;
103 options.timeout_ms = kPermissionTimeoutMs;
104 options.abort = [this]() { return cancelled_ && cancelled_(); };
105 if (!endpoint_->request("session/request_permission", params, result, error, options)) {
106 if (cancelled_ && cancelled_()) {
107 error = "The turn was cancelled";
108 } else {
109 error = "The permission request failed: " + error;
110 }
111 return false;
112 }
113
114 const json::Value* outcome = result.find("outcome");
115 if (outcome == nullptr || !outcome->isObject()) {
116 error = "The permission response has no outcome";
117 return false;
118 }
119 const json::Value* kind = outcome->find("outcome");
120 if (kind == nullptr || !kind->isString()) {
121 error = "The permission response has no outcome kind";
122 return false;
123 }
124 if (kind->string == "cancelled") {
125 error = "The user cancelled the request";
126 return false;
127 }
128 if (kind->string != "selected") {
129 error = "The permission outcome is not understood";
130 return false;
131 }
132 const json::Value* optionId = outcome->find("optionId");
133 if (optionId == nullptr || !optionId->isString()) {
134 error = "The permission response names no option";
135 return false;
136 }
137
138 for (const PermissionChoice& choice : permissionChoices()) {
139 if (choice.id == optionId->string) {
140 answer = choice.label;
141 return true;
142 }
143 }
144 error = "The permission outcome names an option that is not on offer";
145 return false;
146}
