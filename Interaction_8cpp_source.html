18std::string trimmed(const std::string& text) {
19 std::size_t begin = 0;
20 while (begin < text.size() && std::isspace(static_cast<unsigned char>(text[begin])) != 0) {
21 ++begin;
22 }
23 std::size_t end = text.size();
24 while (end > begin && std::isspace(static_cast<unsigned char>(text[end - 1])) != 0) {
25 --end;
26 }
27 return text.substr(begin, end - begin);
28}

30// Parses a one based choice. Returns 0 when the text is not a plain number.
31std::size_t choiceNumber(const std::string& text) {
32 if (text.empty()) {
33 return 0;
34 }
35 std::size_t value = 0;
36 for (const char character : text) {
37 if (std::isdigit(static_cast<unsigned char>(character)) == 0) {
38 return 0;
39 }
40 value = value * 10 + static_cast<std::size_t>(character - '0');
41 if (value > 1000000) {
42 return 0;
43 }
44 }
45 return value;
46}

48bool readOption(const json::Value& arguments, Request& request, std::string& error) {
49 const json::Value* options = arguments.find("options");
50 if (options == nullptr) {
51 return true;
52 }
53 if (!options->isArray()) {
54 error = "request_input options must be an array of strings";
55 return false;
56 }
57 for (const json::Value& option : options->array) {
58 if (!option.isString() || option.string.empty()) {
59 error = "request_input options must be non-empty strings";
60 return false;
61 }
62 request.options.push_back(option.string);
63 }
64 return true;
65}

80bool ConsoleChannel::ask(const Request& request, std::string& answer, std::string& error) {
81 output_ << "\n" << request.question << "\n";
82 if (request.options.empty()) {
83 output_ << "> ";
84 } else {
85 for (std::size_t index = 0; index < request.options.size(); ++index) {
86 output_ << " " << (index + 1) << ". " << request.options[index] << "\n";
87 }
88 output_ << "Choose 1-" << request.options.size() << " or type an option: ";
89 }
90 output_.flush();
91
92 std::string line;
93 if (!std::getline(input_, line)) {
94 error = "Standard input closed before an answer was given.";
95 return false;
96 }
97 line = trimmed(line);
98 if (line.empty()) {
99 error = "No answer was given.";
100 return false;
101 }
102 if (request.options.empty()) {
103 answer = line;
104 return true;
105 }
106
107 const std::size_t choice = choiceNumber(line);
108 if (choice >= 1 && choice <= request.options.size()) {
109 answer = request.options[choice - 1];
110 return true;
111 }
112 for (const std::string& option : request.options) {
113 if (line == option) {
114 answer = option;
115 return true;
116 }
117 }
118 error = "The answer is not one of the offered options.";
119 return false;
120}
