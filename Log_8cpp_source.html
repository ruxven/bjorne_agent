71void block(const std::string& label, const std::string& text, std::size_t limit) {
72 if (g_level < Level::Normal || text.empty()) {
73 return;
74 }
75 std::cerr << "[bjorne_agent " << formatSeconds(elapsedSeconds()) << "] " << label << ":\n";
76 const std::size_t shown = std::min(text.size(), limit);
77 std::size_t pos = 0;
78 while (pos < shown) {
79 std::size_t lineEnd = text.find('\n', pos);
80 if (lineEnd == std::string::npos || lineEnd > shown) {
81 lineEnd = shown;
82 }
83 std::string line = text.substr(pos, lineEnd - pos);
84 // A CRLF from a tool result would otherwise leave a stray carriage return
85 // that overwrites the start of the line on a terminal.
86 if (!line.empty() && line.back() == '\r') {
87 line.pop_back();
88 }
89 std::cerr << " " << line << "\n";
90 pos = lineEnd + 1;
91 }
92 if (shown < text.size()) {
93 std::cerr << " ... [" << (text.size() - shown) << " more characters]\n";
94 }
95 std::cerr.flush();
96}
