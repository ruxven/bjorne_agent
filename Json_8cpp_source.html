164 // Appends the UTF-8 encoding of a Unicode code point.
165 static void appendUtf8(std::string& out, unsigned int code) {
166 if (code < 0x80) {
167 out.push_back(static_cast<char>(code));
168 } else if (code < 0x800) {
169 out.push_back(static_cast<char>(0xC0 | (code >> 6)));
170 out.push_back(static_cast<char>(0x80 | (code & 0x3F)));
171 } else if (code < 0x10000) {
172 out.push_back(static_cast<char>(0xE0 | (code >> 12)));
173 out.push_back(static_cast<char>(0x80 | ((code >> 6) & 0x3F)));
174 out.push_back(static_cast<char>(0x80 | (code & 0x3F)));
175 } else {
176 out.push_back(static_cast<char>(0xF0 | (code >> 18)));
177 out.push_back(static_cast<char>(0x80 | ((code >> 12) & 0x3F)));
178 out.push_back(static_cast<char>(0x80 | ((code >> 6) & 0x3F)));
179 out.push_back(static_cast<char>(0x80 | (code & 0x3F)));

183 // Reads four hex digits at pos_ and advances past them.
184 unsigned int readHex4() {
185 if (pos_ + 4 > text_.size()) {
186 throw std::runtime_error("Truncated JSON unicode escape");
187 }
188 unsigned int code = 0;
189 for (size_t i = 0; i < 4; ++i) {
190 const char digit = text_[pos_ + i];
191 code <<= 4;
192 if (digit >= '0' && digit <= '9') {
193 code |= static_cast<unsigned int>(digit - '0');
194 } else if (digit >= 'a' && digit <= 'f') {
195 code |= static_cast<unsigned int>(digit - 'a' + 10);
196 } else if (digit >= 'A' && digit <= 'F') {
197 code |= static_cast<unsigned int>(digit - 'A' + 10);
198 } else {
199 throw std::runtime_error("Invalid JSON unicode escape");
200 }
201 }
202 pos_ += 4;
203 return code;
204 }

206 // Handles \uXXXX with pos_ on the 'u'. A high surrogate must be followed by
207 // a low surrogate, which is how models send characters outside the basic
208 // multilingual plane, emoji included.
209 void parseUnicodeEscape(std::string& out) {
210 ++pos_;
211 unsigned int code = readHex4();
212 if (code >= 0xD800 && code <= 0xDBFF) {
213 if (pos_ + 6 > text_.size() || text_[pos_] != '\\' || text_[pos_ + 1] != 'u') {
214 throw std::runtime_error("Lone high surrogate in JSON string");
215 }
216 pos_ += 2;
217 const unsigned int low = readHex4();
218 if (low < 0xDC00 || low > 0xDFFF) {
219 throw std::runtime_error("Invalid low surrogate in JSON string");
220 }
221 code = 0x10000 + ((code - 0xD800) << 10) + (low - 0xDC00);
222 } else if (code >= 0xDC00 && code <= 0xDFFF) {
223 throw std::runtime_error("Lone low surrogate in JSON string");
224 }
225 appendUtf8(out, code);
226 }

228 std::string parseString() {
229 ++pos_;
230 std::string result;
231 while (true) {
232 if (pos_ >= text_.size()) {
233 throw std::runtime_error("Unterminated JSON string");
234 }
235 const char c = text_[pos_];
236 if (c == '"') {
237 ++pos_;
238 break;
239 }
240 if (static_cast<unsigned char>(c) < 0x20) {
241 throw std::runtime_error("Control character in JSON string");
242 }
243 if (c == '\\') {
244 ++pos_;
245 const char esc = peek();
246 switch (esc) {
247 case '"': result.push_back('"'); ++pos_; break;
248 case '\\': result.push_back('\\'); ++pos_; break;
249 case '/': result.push_back('/'); ++pos_; break;
250 case 'b': result.push_back('\b'); ++pos_; break;
251 case 'f': result.push_back('\f'); ++pos_; break;
252 case 'n': result.push_back('\n'); ++pos_; break;
253 case 'r': result.push_back('\r'); ++pos_; break;
254 case 't': result.push_back('\t'); ++pos_; break;
255 case 'u': {
256 parseUnicodeEscape(result);
257 break;
258 }
259 default:
260 throw std::runtime_error("Invalid escape in JSON string");
261 }
262 continue;
263 }
264 result.push_back(c);
265 ++pos_;
266 }
267 return result;

289void writeString(const std::string& text, std::string& out) {
290 static const char kHex[] = "0123456789abcdef";
291 out.push_back('"');
292 for (const char c : text) {
293 const unsigned char byte = static_cast<unsigned char>(c);
294 switch (c) {
295 case '"': out += "\\\""; break;
296 case '\\': out += "\\\\"; break;
297 case '\b': out += "\\b"; break;
298 case '\f': out += "\\f"; break;
299 case '\n': out += "\\n"; break;
300 case '\r': out += "\\r"; break;
301 case '\t': out += "\\t"; break;
302 default:
303 if (byte < 0x20) {
304 out += "\\u00";
305 out.push_back(kHex[byte >> 4]);
306 out.push_back(kHex[byte & 0x0F]);
307 } else {
308 out.push_back(c);
309 }
310 break;
311 }
312 }
313 out.push_back('"');
314}

316void writeNumber(double number, std::string& out) {
317 if (!std::isfinite(number)) {
318 throw std::runtime_error("Cannot serialize a non-finite JSON number");
319 }
320 // Print an integral value without a decimal point. The range keeps the
321 // conversion to long long well defined; larger values fall through to the
322 // general format, which still produces valid JSON.
323 if (number == std::floor(number) && std::fabs(number) < 9.0e15) {
324 char buffer[32];
325 const int written = std::snprintf(buffer, sizeof(buffer), "%lld", static_cast<long long>(number));
326 if (written < 0 || static_cast<size_t>(written) >= sizeof(buffer)) {
327 throw std::runtime_error("Cannot format a JSON number");
328 }
329 out += buffer;
330 return;
331 }
332 char buffer[32];
333 // %.17g round-trips every double, so parsing the text returns the same value.
334 const int written = std::snprintf(buffer, sizeof(buffer), "%.17g", number);
335 if (written < 0 || static_cast<size_t>(written) >= sizeof(buffer)) {
336 throw std::runtime_error("Cannot format a JSON number");
337 }
338 out += buffer;
339}
