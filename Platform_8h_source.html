39// Winsock needs a process wide startup before the first socket call. The first
40// call starts it once; later calls reuse the result. A build that fails to
41// start reports it, and the caller turns that into a connection error.
42inline bool initializeSockets() {
43 static const bool ready = [] {
44 WSADATA data{};
45 return ::WSAStartup(MAKEWORD(2, 2), &data) == 0;
46 }();
47 return ready;
48}

54// Reads a variable. A variable that is unset and one that holds an empty value
55// both report false, which keeps the two platforms in step.
56inline bool getEnvironmentVariable(const std::string& name, std::string& out) {
57 const DWORD size = ::GetEnvironmentVariableA(name.c_str(), nullptr, 0);
58 if (size == 0) {
59 return false;
60 }
61 std::string buffer(size, '\0');
62 const DWORD written = ::GetEnvironmentVariableA(name.c_str(), buffer.data(), size);
63 if (written == 0 || written >= size) {
64 return false;
65 }
66 out.assign(buffer.data(), written);
67 return true;
68}
