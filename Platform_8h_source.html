41// Winsock needs a process wide startup before the first socket call. The first
42// call starts it once; later calls reuse the result. A build that fails to
43// start reports it, and the caller turns that into a connection error.
44inline bool initializeSockets() {
45 static const bool ready = [] {
46 WSADATA data{};
47 return ::WSAStartup(MAKEWORD(2, 2), &data) == 0;
48 }();
49 return ready;
50}

56// Reads a variable. A variable that is unset and one that holds an empty value
57// both report false, which keeps the two platforms in step.
58inline bool getEnvironmentVariable(const std::string& name, std::string& out) {
59 const DWORD size = ::GetEnvironmentVariableA(name.c_str(), nullptr, 0);
60 if (size == 0) {
61 return false;
62 }
63 std::string buffer(size, '\0');
64 const DWORD written = ::GetEnvironmentVariableA(name.c_str(), buffer.data(), size);
65 if (written == 0 || written >= size) {
66 return false;
67 }
68 out.assign(buffer.data(), written);
69 return true;
70}
