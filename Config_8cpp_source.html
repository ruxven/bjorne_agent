124bool load(const std::string& path, Config& out, std::string& error) {
125 std::ifstream file(path);
126 if (!file.is_open()) {
127 error = "Cannot open config file: " + path;
128 return false;
129 }
130 std::stringstream buffer;
131 buffer << file.rdbuf();
132 json::Value root;
133 try {
134 root = json::parse(buffer.str());
135 } catch (const std::exception& e) {
136 error = std::string("JSON parse error: ") + e.what();
137 return false;
138 }
139 if (!root.isObject()) {
140 error = "Config root must be a JSON object";
141 return false;
142 }
143
144 const json::Value* version = root.find("version");
145 if (version == nullptr || !version->isNumber()) {
146 error = "Config version must be a number";
147 return false;
148 }
149 out.version = static_cast<long long>(version->number);
150 if (out.version != 1) {
151 error = "Unsupported config version: " + std::to_string(out.version);
152 return false;
153 }
154
155 loadAgent(root, out.agent);
156 loadModels(root, out.models);
157 loadMcpServers(root, out.mcp_servers);
158 return true;
159}
