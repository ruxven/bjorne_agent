65// Renders one content item. Text passes through; any other type becomes a
66// short placeholder, because a text only agent cannot display an image.
67std::string contentText(const json::Value& item) {
68 if (item.isString()) {
69 return item.string;
70 }
71 if (!item.isObject()) {
72 return std::string();
73 }
74 const json::Value* type = item.find("type");
75 if (type != nullptr && type->isString() && type->string == "text") {
76 const json::Value* text = item.find("text");
77 if (text != nullptr && text->isString()) {
78 return text->string;
79 }
80 }
81 std::string label = "non-text content";
82 if (type != nullptr && type->isString() && !type->string.empty()) {
83 label = type->string + " content";
84 }
85 return "[" + label + " omitted]";
86}

95 // A server may send requests of its own. Ping is the only one this client
96 // promises, and anything else gets a method-not-found response rather than
97 // a silent drop.
98 endpoint_->setRequestHandler([](const std::string& method, const json::Value& params, json::Value& result,
99 jsonrpc::Error& error) {
100 (void)params;
101 if (method == "ping") {
102 result = objectValue();
103 return true;
104 }
105 error.code = jsonrpc::kMethodNotFound;
106 error.message = "Method not found: " + method;
107 return false;
108 });

111bool Client::initialize(std::string& error) {
112 json::Value params = objectValue();
113 params.object["protocolVersion"] = stringValue(kProtocolVersion);
114 params.object["capabilities"] = objectValue();
115 json::Value client = objectValue();
116 client.object["name"] = stringValue("bjorne_agent");
117 client.object["version"] = stringValue(kClientVersion);
118 params.object["clientInfo"] = client;
119
120 json::Value result;
121 if (!endpoint_->request("initialize", params, result, error, kHandshakeTimeoutMs)) {
122 return false;
123 }
124 if (!result.isObject()) {
125 error = "The initialize response is not an object";
126 return false;
127 }
128 return endpoint_->notify("notifications/initialized", objectValue(), error);
129}

151void Client::appendTools(const json::Value* list, std::vector<ToolInfo>& tools) {
152 if (list == nullptr || !list->isArray()) {
153 return;
154 }
155 for (const json::Value& entry : list->array) {
156 if (!entry.isObject()) {
157 continue;
158 }
159 const json::Value* name = entry.find("name");
160 if (name == nullptr || !name->isString() || name->string.empty()) {
161 continue;
162 }
163 ToolInfo tool;
164 tool.name = name->string;
165 const json::Value* description = entry.find("description");
166 if (description != nullptr && description->isString()) {
167 tool.description = description->string;
168 }
169 const json::Value* schema = entry.find("inputSchema");
170 if (schema != nullptr && schema->isObject()) {
171 tool.input_schema = *schema;
172 } else {
173 tool.input_schema = objectValue();
174 }
175 tools.push_back(std::move(tool));
176 }
177}

179bool Client::listTools(std::vector<ToolInfo>& tools, std::string& error) {
180 tools.clear();
181 std::string cursor;
182 for (int page = 0; page < kMaxToolPages; ++page) {
183 json::Value result;
184 if (!fetchToolPage(cursor, result, error)) {
185 return false;
186 }
187 appendTools(result.find("tools"), tools);
188
189 const json::Value* next = result.find("nextCursor");
190 if (next == nullptr || !next->isString() || next->string.empty()) {
191 return true;
192 }
193 cursor = next->string;
194 }
195 error = "The server returned more than " + std::to_string(kMaxToolPages) + " pages of tools";
196 return false;
197}

199bool Client::callTool(const std::string& name, const json::Value& arguments, std::string& output, bool& is_error,
200 std::string& error) {
201 json::Value params = objectValue();
202 params.object["name"] = stringValue(name);
203 params.object["arguments"] = arguments;
204
205 json::Value result;
206 if (!endpoint_->request("tools/call", params, result, error, kCallTimeoutMs)) {
207 return false;
208 }
209 if (!result.isObject()) {
210 error = "The tools/call response is not an object";
211 return false;
212 }
213
214 output.clear();
215 const json::Value* content = result.find("content");
216 if (content != nullptr && content->isArray()) {
217 bool first = true;
218 for (const json::Value& item : content->array) {
219 const std::string text = contentText(item);
220 if (text.empty()) {
221 continue;
222 }
223 if (!first) {
224 output.push_back('\n');
225 }
226 output.append(text);
227 first = false;
228 }
229 }
230
231 is_error = false;
232 const json::Value* flag = result.find("isError");
233 if (flag != nullptr && flag->isBoolean()) {
234 is_error = flag->boolean;
235 }
236 return true;
237}

286tools::ToolResult Server::call(const std::string& tool, const std::string& arguments) {
287 json::Value parsed;
288 if (!tools::parseToolArguments(arguments, parsed)) {
289 return tools::ToolResult{false, "The tool arguments are not a JSON object"};
290 }
291
292 const auto attempt = [this, &tool, &parsed]() {
293 std::string output;
294 bool is_error = false;
295 std::string error;
296 if (!client_ || client_->closed()) {
297 return tools::ToolResult{false, "The MCP server " + name_ + " is not running"};
298 }
299 if (!client_->callTool(tool, parsed, output, is_error, error)) {
300 return tools::ToolResult{false, error};
301 }
302 if (is_error) {
303 return tools::ToolResult{false, output.empty() ? "The tool reported an error" : output};
304 }
305 return tools::ToolResult{true, output};
306 };
307
308 tools::ToolResult result = attempt();
309 if (result.ok) {
310 return result;
311 }
312
313 // A transport failure may mean the server died. One restart and one retry
314 // keeps a crash to a lost call rather than a lost run. A tool that merely
315 // reported an error is not retried, because the server itself worked.
316 const bool may_restart = !client_ || client_->closed() || result.output.find("timed out") != std::string::npos;
317 if (!may_restart) {
318 return result;
319 }
320
321 log::fail("MCP server " + key() + " failed; restarting it");
322 shutdown();
323 std::string error;
324 if (!launch(error) || !initializeConnection(error)) {
325 shutdown();
326 return tools::ToolResult{false, "MCP server " + key() + " could not be restarted: " + error};
327 }
328 return attempt();
329}

331void registerServers(const std::map<std::string, config::McpServerConfig>& servers,
332 tools::ToolRegistry& registry) {
333 for (const auto& [name, config] : servers) {
334 if (config.command.empty()) {
335 log::fail("MCP server " + name + " has no command");
336 continue;
337 }
338 auto server = std::make_shared<Server>(name, config);
339 std::string error;
340 if (!server->start(error)) {
341 log::fail("MCP server " + name + " is unavailable: " + error);
342 continue;
343 }
344 log::detail("MCP server " + name + ": " + std::to_string(server->tools().size()) + " tool(s)");
345 for (const ToolInfo& tool : server->tools()) {
346 const std::string exposed = toolName(name, tool.name);
347 tools::ToolDefinition definition;
348 definition.name = exposed;
349 definition.description = tool.description;
350 definition.parameters = tool.input_schema;
351 if (!definition.parameters.isObject()) {
352 definition.parameters = objectValue();
353 }
354 registry.registerTool(definition, [server, tool_name = tool.name](const std::string& arguments) {
355 return server->call(tool_name, arguments);
356 });
357 log::detail("MCP tool: " + exposed);
358 }
359 }
