21std::string escapeJson(const std::string& text) {
22 std::string result;
23 for (const char c : text) {
24 switch (c) {
25 case '"': result += "\\\""; break;
26 case '\\': result += "\\\\"; break;
27 case '\n': result += "\\n"; break;
28 case '\r': result += "\\r"; break;
29 case '\t': result += "\\t"; break;
30 default:
31 if (static_cast<unsigned char>(c) < 0x20) {
32 const char hex[] = "0123456789abcdef";
33 const unsigned int value = static_cast<unsigned char>(c);
34 result += "\\u00";
35 result += hex[(value >> 4) & 0xf];
36 result += hex[value & 0xf];
37 } else {
38 result += c;
39 }
40 }
41 }
42 return result;
43}

89void parseToolCalls(const json::Value& message, std::vector<ToolCall>& out) {
90 const json::Value* toolCalls = message.find("tool_calls");
91 if (toolCalls == nullptr || !toolCalls->isArray()) {
92 return;
93 }
94 for (const json::Value& call : toolCalls->array) {
95 if (!call.isObject()) {
96 continue;
97 }
98 ToolCall parsed;
99 const json::Value* id = call.find("id");
100 if (id != nullptr && id->isString()) {
101 parsed.id = id->string;
102 }
103 const json::Value* function = call.find("function");
104 if (function != nullptr && function->isObject()) {
105 const json::Value* name = function->find("name");
106 const json::Value* arguments = function->find("arguments");
107 if (name != nullptr && name->isString()) {
108 parsed.name = name->string;
109 }
110 if (arguments != nullptr && arguments->isString()) {
111 parsed.arguments = arguments->string;
112 }
113 }
114 if (!parsed.name.empty()) {
115 out.push_back(parsed);
116 }
117 }

120// Extracts the first choice's message from a chat completions body.
121bool parseChatResponse(const std::string& body, ModelResponse& out, std::string& error) {
122 json::Value root;
123 try {
124 root = json::parse(body);
125 } catch (const std::exception& e) {
126 error = std::string("Model response JSON parse error: ") + e.what();
127 return false;
128 }
129 const json::Value* choices = root.find("choices");
130 if (choices == nullptr || !choices->isArray() || choices->array.empty()) {
131 error = "Model response missing choices";
132 return false;
133 }
134 const json::Value& choice = choices->array.front();
135 const auto messageIt = choice.object.find("message");
136 if (messageIt == choice.object.end()) {
137 error = "Model response missing message";
138 return false;
139 }
140 const json::Value& message = messageIt->second;
141 const json::Value* content = message.find("content");
142 if (content != nullptr && content->isString()) {
143 out.content = content->string;
144 }
145 parseToolCalls(message, out.tool_calls);
146 return true;
147}

163bool ModelClient::respond(const std::vector<ModelMessage>& messages, ModelResponse& out, std::string& error) {
164 http::Url url;
165 if (!http::parseUrl(endpoint_, url, error)) {
166 return false;
167 }
168
169 std::string apiKey;
170 if (!platform::getEnvironmentVariable(apiKeyEnv_, apiKey)) {
171 error = "Missing API key environment variable: " + apiKeyEnv_;
172 log::fail(error);
173 return false;
174 }
175 log::detail("API key found in " + apiKeyEnv_);
176
177 std::unique_ptr<http::HttpClient> client;
178 if (transport_ != nullptr) {
179 client = std::make_unique<http::HttpClient>(transport_);
180 } else if (url.tls) {
181#if BJORNE_AGENT_HAS_OPENSSL
182 http::OpenSslTransport::Options tlsOptions;
183 tlsOptions.ignore_self_signed = ignoreSelfSigned_;
184 tlsOptions.hostname = hostname_;
185 client = http::HttpClient::createOpenSsl(tlsOptions);
186#else
187 error = "Endpoint " + endpoint_ + " uses https, but this build has no TLS support";
188 return false;
189#endif
190 } else {
191 client = http::HttpClient::createPlainTcp();
192 }
193
194 http::HttpRequest request;
195 request.host = url.host;
196 request.port = url.port;
197 request.path = url.path + "/chat/completions";
198 request.body = requestBodyJson(model_, messages, toolsJson_);
199 request.headers = {{ "Authorization", "Bearer " + apiKey }};
200
201 http::HttpResponse response;
202 log::progress("Requesting " + url.scheme + "://" + url.host + ":" + url.port + request.path);
203 if (!client->request(request, response, error)) {
204 log::fail(error);
205 return false;
206 }
207 log::detail("Model endpoint returned status " + std::to_string(response.status) + " with " +
208 std::to_string(response.body.size()) + " bytes");
209 if (response.status < 200 || response.status >= 300) {
210 error = "Model endpoint returned status " + std::to_string(response.status) + ": " + response.body;
211 log::fail(error);
212 return false;
213 }
214
215 ModelResponse result;
216 if (!parseChatResponse(response.body, result, error)) {
217 log::fail(error);
218 return false;
219 }
220 out = result;
221 return true;
222}
